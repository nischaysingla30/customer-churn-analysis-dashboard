Pandas & NumPy Cleaning Pipeline

Clean raw incoming transactional data: standardizing inconsistent formats, handling missing inputs, and preparing arrays for dashboard serialization.

data_cleaning.py
# data_cleaning.py
import pandas as pd
import numpy as np

def clean_churn_data(input_path="customer_churn_raw.csv"):
    # 1. Load raw dataset containing missing & formatting issues
    df = pd.read_csv(input_path)
    
    # 2. Standardize Categorical Casing (Gender values like 'female'/'FEMALE')
    df['gender'] = df['gender'].str.capitalize()
    
    # 3. Handle TotalCharges columns (strip white space & replace with NaN)
    df['TotalCharges'] = df['TotalCharges'].replace(r'^\s*$', np.nan, regex=True)
    df['TotalCharges'] = pd.to_numeric(df['TotalCharges'], errors='coerce')
    
    # 4. Impute missing charges values
    # Case A: new customer (tenure = 0), set TotalCharges = 0.0
    df.loc[(df['TotalCharges'].isnull()) & (df['tenure'] == 0), 'TotalCharges'] = 0.0
    
    # Case B: existing customer missing charges, impute: tenure * MonthlyCharges
    missing_mask = (df['TotalCharges'].isnull()) & (df['tenure'] > 0)
    df.loc[missing_mask, 'TotalCharges'] = round(df['tenure'] * df['MonthlyCharges'], 2)
    
    # 5. Remove duplicates based on unique Customer IDs
    df.drop_duplicates(subset=['customerID'], keep='first', inplace=True)
    
    # 6. Save formatted file
    df.to_csv("customer_churn_clean.csv", index=False)
    print("ETL Pipeline completed successfully.")
